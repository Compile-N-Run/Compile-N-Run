{{- /* Resolve a roadmap node (data/roadmap.yaml) to {title, href, prefix, total, note, kind, tut, sections}. */ -}} {{- $n := . -}} {{- $r := dict "title" ($n.title | default "") "note" ($n.note | default "") "href" "" "prefix" "" "total" 0 "kind" "missing" "tut" "" "sections" slice -}} {{- $path := "" -}}{{- $kind := "" -}} {{- with $n.tut }}{{ $path = . }}{{ $kind = "tut" }}{{ end -}} {{- with $n.section }}{{ $path = . }}{{ $kind = "section" }}{{ end -}} {{- with $n.page }}{{ $path = . }}{{ $kind = "page" }}{{ end -}} {{- with $n.track }} {{- $t := index (where site.Data.roadmap.tracks "id" .) 0 -}} {{- $r = merge $r (dict "kind" "track" "href" (printf "#%s" .) "title" (printf "%s roadmap" $t.title)) -}} {{- end -}} {{- if $path -}} {{- $p := site.GetPage (printf "/docs/%s" $path) -}} {{- if not $p }}{{ errorf "roadmap: no page for %q" $path }}{{ end -}} {{- $prefix := printf "/docs/%s/" (replaceRE `(^|/)\d+-` "$1" $path) -}} {{- if eq $kind "page" -}} {{- $r = merge $r (dict "kind" "page" "href" $p.RelPermalink "prefix" $p.RelPermalink "total" 1 "title" ($r.title | default ($p.Params.sidebar_label | default $p.Title))) -}} {{- else -}} {{- $flat := partial "flat.html" $p -}} {{- $href := $p.RelPermalink -}} {{- if not $href }}{{ with index $flat 0 }}{{ $href = .RelPermalink }}{{ end }}{{ end -}} {{- $title := $r.title | default $p.Title -}} {{- if eq $kind "tut" -}} {{- with index (where site.Data.tutorials "dir" $path) 0 -}} {{- $href = .url -}} {{- if not $r.title }}{{ $title = replaceRE ` Tutorial$` "" .label }}{{ end -}} {{- end -}} {{- $secs := slice -}} {{- range $p.Pages.ByWeight -}} {{- $label := .Params.sidebar_label | default .Title -}} {{- if .IsSection -}} {{- $f := partial "flat.html" . -}} {{- $h := .RelPermalink -}} {{- if not $h }}{{ with index $f 0 }}{{ $h = .RelPermalink }}{{ end }}{{ end -}} {{- $sp := printf "/docs/%s/" (replaceRE `(^|/)\d+-` "$1" (strings.TrimSuffix "/" (strings.TrimPrefix "docs/" .File.Dir))) -}} {{- $secs = $secs | append (dict "title" $label "href" $h "prefix" $sp "total" (len $f)) -}} {{- else -}} {{- $secs = $secs | append (dict "title" $label "href" .RelPermalink "prefix" .RelPermalink "total" 1) -}} {{- end -}} {{- end -}} {{- $r = merge $r (dict "sections" $secs "tut" $path) -}} {{- end -}} {{- $r = merge $r (dict "kind" $kind "href" $href "prefix" $prefix "total" (len $flat) "title" $title) -}} {{- end -}} {{- end -}} {{- return $r -}}
