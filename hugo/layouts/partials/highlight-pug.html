{{- /* Chroma has no Pug lexer: tokenize Pug line by line into chroma's markup and classes. Context: the source string. */ -}} {{- $keywords := slice "doctype" "if" "else" "unless" "each" "for" "while" "case" "when" "default" "include" "extends" "block" "append" "prepend" "mixin" "yield" -}} {{- $out := slice -}} {{- $textIndent := -1 -}}{{- /* Indent of a `tag.` line whose children are plain text. */ -}} {{- range split (strings.TrimSuffix "\n" .) "\n" -}} {{- $m := index (findRESubmatch `^(\s*)(.*)$` .) 0 -}} {{- $ws := index $m 1 -}}{{- $rest := index $m 2 -}} {{- $indent := len $ws -}} {{- $html := "" -}} {{- if and (ge $textIndent 0) (or (gt $indent $textIndent) (eq $rest "")) -}} {{- $html = htmlEscape $rest -}} {{- else -}} {{- $textIndent = -1 -}} {{- if hasPrefix $rest "//" -}} {{- $html = printf `%s` (htmlEscape $rest) -}} {{- else if hasPrefix $rest "|" -}} {{- $html = printf `|%s` (partial "highlight-pug-text" (substr $rest 1)) -}} {{- else if hasPrefix $rest "-" -}} {{- $html = printf `-%s` (partial "highlight-pug-text" (substr $rest 1)) -}} {{- else if hasPrefix $rest "+" -}} {{- $t := index (findRESubmatch `^\+([\w-]+)(.*)$` $rest) 0 -}} {{- if $t -}} {{- $html = printf `+%s%s` (index $t 1) (partial "highlight-pug-text" (index $t 2)) -}} {{- else -}} {{- $html = htmlEscape $rest -}} {{- end -}} {{- else -}} {{- $t := index (findRESubmatch `^([A-Za-z][\w:-]*)?((?:[.#][\w-]+)*)(\([^)]*\))?(!?=|:|\.$)?(.*)$` $rest) 0 -}} {{- $tag := index $t 1 -}}{{- $sel := index $t 2 -}}{{- $attrs := index $t 3 -}}{{- $op := index $t 4 -}}{{- $tail := index $t 5 -}} {{- if in $keywords $tag -}} {{- $html = printf `%s%s%s%s%s` $tag (htmlEscape $sel) (partial "highlight-pug-attrs" $attrs) (htmlEscape $op) (partial "highlight-pug-text" $tail) -}} {{- else if or $tag $sel -}} {{- with $tag }}{{ $html = printf `%s` . }}{{ end -}} {{- range findRE `[.#][\w-]+` $sel -}} {{- $html = printf `%s%s` $html (cond (hasPrefix . ".") "nc" "nn") . -}} {{- end -}} {{- $html = printf `%s%s` $html (partial "highlight-pug-attrs" $attrs) -}} {{- with $op }}{{ $html = printf `%s%s` $html . }}{{ end -}} {{- $html = printf `%s%s` $html (partial "highlight-pug-text" $tail) -}} {{- if eq $op "." }}{{ $textIndent = $indent }}{{ end -}} {{- else -}} {{- $html = partial "highlight-pug-text" $rest -}} {{- end -}} {{- end -}} {{- end -}} {{- $out = $out | append (printf `%s%s ` $ws $html) -}} {{- end -}} {{- return (delimit $out "") -}}
